Task 1

SELECT SUM(products.price * line_items.quantity) as total_price, line_items.order_id
FROM Products
Inner Join line_items 
ON products.product_id=line_items.product_id
Inner Join orders
ON line_items.order_id = orders.order_id
GROUP BY line_items.order_id
ORDER BY line_items.order_id
LIMIT 5;

Task 2
SELECT c.customer_name, AVG(t.total_price) as average_order_price
FROM customers c
JOIN (
	SELECT SUM(products.price * line_items.quantity) as total_price, orders.customer_id
	FROM Products
	Inner Join line_items 
	ON products.product_id=line_items.product_id
	Inner Join orders
	ON line_items.order_id = orders.order_id
	Inner Join customers
	ON orders.customer_id = customers.customer_id
	GROUP BY line_items.order_id, orders.customer_id
) t
ON c.customer_id = t.customer_id
GROUP BY c.customer_id, customer_name
ORDER BY customer_name

Select Count(Orders.order_id) As order_count, employees.first_name, employees.last_name
FROM Orders
INNER JOIN Employees ON Orders.employee_id = Employees.employee_id
Group By Employees.employee_id
HAVING COUNT(Orders.order_id) > 5
Order By last_name;
